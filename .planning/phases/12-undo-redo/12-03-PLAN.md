2. Implement handleUndo():
```go
func (m Model) handleUndo() (tea.Model, tea.Cmd) {
    // Flush pending edits to document (CRITICAL - Pitfall 4)
    m.transitionToProcessing()

    // Commit any pending batch before undoing
    m.undoManager.CommitCurrentBatch()

    // Get batch to undo
    batch, ok := m.undoManager.Undo()
    if !ok {
        m.statusMsg = "Nothing to undo"
        return m, nil
    }

    // Apply operations in reverse order
    for i := len(batch.Operations) - 1; i >= 0; i-- {
        m.applyOperationReverse(batch.Operations[i])
    }

    // Restore cursor from first operation (chronologically last - has pre-batch state)
    if len(batch.Operations) > 0 {
        op := batch.Operations[0]
        m.cursorLine = op.CursorLine
        m.cursorCol = op.CursorCol
        m.scrollOffset = op.ScrollOffset
    }

    // Re-evaluate document
    m.redetectBlockTypes()
    m.reEvaluate()

    m.statusMsg = "Undo"
    m.modified = true
    return m, nil
}
```

3. Implement handleRedo() similarly but using batch.Operations in forward order

4. Implement applyOperationReverse():
   - For OpInsert: Delete the NewText at position
   - For OpDelete: Insert the OldText at position
   - For OpReplace: Replace NewText with OldText

5. Implement applyOperationForward() for redo:
   - For OpInsert: Insert the NewText at position
   - For OpDelete: Delete OldText.length chars at position
   - For OpReplace: Replace OldText with NewText

Key insight: Operations store the position and text, so applying them is direct string manipulation on the document.
