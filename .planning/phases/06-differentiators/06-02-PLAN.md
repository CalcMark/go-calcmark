```go
// FunctionSuggestionSource provides function name suggestions.
type FunctionSuggestionSource struct{}

func NewFunctionSuggestionSource() *FunctionSuggestionSource {
    return &FunctionSuggestionSource{}
}

func (f *FunctionSuggestionSource) GetSuggestions(prefix string) []components.Suggestion {
    prefix = strings.ToLower(prefix)
    var suggestions []components.Suggestion

    for _, fn := range interpreter.BuiltinFunctions {
        // Match primary name or any synonym
        matched := strings.HasPrefix(strings.ToLower(fn.Name), prefix)
        matchedSynonym := ""
        for _, syn := range fn.Synonyms {
            if strings.HasPrefix(strings.ToLower(syn), prefix) {
                matched = true
                matchedSynonym = syn
                break
            }
        }

        if matched {
            // Format name with synonyms for display
            name := fn.Name
            if len(fn.Synonyms) > 0 {
                name = fmt.Sprintf("%s (%s)", fn.Name, strings.Join(fn.Synonyms, ", "))
            }

            suggestions = append(suggestions, components.Suggestion{
                Name:        name,
                Category:    fn.Category,
                Description: fn.Description,
                Syntax:      fn.Signature,
                // Store the actual insertion text separately
                InsertText:  fn.Name,
            })
        }
    }

    return suggestions
}
```

```go
// UnitSuggestionSource provides unit name suggestions.
type UnitSuggestionSource struct{}

func NewUnitSuggestionSource() *UnitSuggestionSource {
    return &UnitSuggestionSource{}
}

func (u *UnitSuggestionSource) GetSuggestions(prefix string) []components.Suggestion {
    prefix = strings.ToLower(prefix)
    var suggestions []components.Suggestion

    for _, unit := range units.StandardUnits {
        matched := strings.HasPrefix(strings.ToLower(unit.Canonical), prefix)
        if !matched {
            matched = strings.HasPrefix(strings.ToLower(unit.Symbol), prefix)
        }
        // Also check aliases
        for _, alias := range unit.Aliases {
            if strings.HasPrefix(strings.ToLower(alias), prefix) {
                matched = true
                break
            }
        }

        if matched {
            suggestions = append(suggestions, components.Suggestion{
                Name:        unit.Canonical,
                Category:    unit.Quantity,
                Description: unit.Description,
                Syntax:      unit.Symbol,
                InsertText:  unit.Canonical,
            })
        }
    }

    return suggestions
}
```

```go
// VariableSuggestionSource provides variable name suggestions from current document.
type VariableSuggestionSource struct {
    getVariables func() map[string]string // Returns varName -> formattedValue
}

func NewVariableSuggestionSource(getVars func() map[string]string) *VariableSuggestionSource {
    return &VariableSuggestionSource{getVariables: getVars}
}

func (v *VariableSuggestionSource) GetSuggestions(prefix string) []components.Suggestion {
    prefix = strings.ToLower(prefix)
    var suggestions []components.Suggestion

    for varName, value := range v.getVariables() {
        if strings.HasPrefix(strings.ToLower(varName), prefix) {
            suggestions = append(suggestions, components.Suggestion{
                Name:        varName,
                Category:    "variable",
                Description: value,
                InsertText:  varName,
            })
        }
    }

    return suggestions
}
```

```go
// CombinedSuggestionSource queries multiple sources and merges results.
type CombinedSuggestionSource struct {
    sources []components.SuggestionSource
}

func NewCombinedSuggestionSource(sources ...components.SuggestionSource) *CombinedSuggestionSource {
    return &CombinedSuggestionSource{sources: sources}
}

func (c *CombinedSuggestionSource) GetSuggestions(prefix string) []components.Suggestion {
    var all []components.Suggestion
    for _, src := range c.sources {
        all = append(all, src.GetSuggestions(prefix)...)
    }
    // Sort: functions first, then units, then variables
    // Within each category, sort alphabetically
    sort.Slice(all, func(i, j int) bool {
        catOrder := map[string]int{"Math": 0, "Conversion": 1, "Network": 2, "Storage": 3, "Capacity": 4, "variable": 10}
        ci, cj := catOrder[all[i].Category], catOrder[all[j].Category]
        if ci != cj {
            return ci < cj
        }
        return all[i].Name < all[j].Name
    })
    return all
}
```

5. Add autocomplete key handlers:
```go
// handleTabKey handles Tab for autocomplete trigger or acceptance
func (m Model) handleTabKey() (tea.Model, tea.Cmd) {
    if m.mode == StateAutocomplete {
        // Accept current selection
        return m.acceptAutocomplete()
    }
    // Trigger autocomplete
    return m.triggerAutocomplete()
}

// triggerAutocomplete initiates autocomplete mode
func (m Model) triggerAutocomplete() (tea.Model, tea.Cmd) {
    // Extract word prefix at cursor position
    prefix := m.getCurrentWordPrefix()
    if prefix == "" {
        return m, nil
    }

    suggestions := m.suggestionSource.GetSuggestions(prefix)
    if len(suggestions) == 0 {
        return m, nil
    }

    m.mode = StateAutocomplete
    m.autocompleteState = components.AutosuggestState{
        Suggestions: suggestions,
        Selected:    0,
        Visible:     true,
        Prefix:      prefix,
    }
    return m, nil
}

// getCurrentWordPrefix extracts the word being typed at cursor
func (m *Model) getCurrentWordPrefix() string {
    m.loadCurrentLineIntoEditBuffer()
    if m.cursorCol == 0 {
        return ""
    }

    // Walk backwards to find word start
    start := m.cursorCol
    for start > 0 {
        ch := m.editBuf[start-1]
        if !isWordChar(ch) {
            break
        }
        start--
    }

    if start >= m.cursorCol {
        return ""
    }
    return m.editBuf[start:m.cursorCol]
}

func isWordChar(ch byte) bool {
    return (ch >= 'a' && ch <= 'z') ||
           (ch >= 'A' && ch <= 'Z') ||
           (ch >= '0' && ch <= '9') ||
           ch == '_'
}

// acceptAutocomplete inserts the selected suggestion
func (m Model) acceptAutocomplete() (tea.Model, tea.Cmd) {
    if m.autocompleteState.Selected < 0 ||
       m.autocompleteState.Selected >= len(m.autocompleteState.Suggestions) {
        m.mode = StateDefault
        return m, nil
    }

    selected := m.autocompleteState.Suggestions[m.autocompleteState.Selected]
    insertText := selected.InsertText
    if insertText == "" {
        insertText = selected.Name
    }

    // Replace prefix with selected suggestion
    prefix := m.autocompleteState.Prefix
    prefixStart := m.cursorCol - len(prefix)
    if prefixStart < 0 {
        prefixStart = 0
    }

    m.editBuf = m.editBuf[:prefixStart] + insertText + m.editBuf[m.cursorCol:]
    m.cursorCol = prefixStart + len(insertText)

    m.mode = StateDefault
    m.autocompleteState = components.AutosuggestState{}
    m.transitionToEditing()
    return m.debounceUpdate()
}
```

```go
// insertAutocompleteDropdown inserts the dropdown into the UI at the cursor position.
func insertAutocompleteDropdown(lines []string, dropdown string, cursorLine, scrollOffset int) []string {
    // Calculate visual position for dropdown (below cursor line)
    visualLine := cursorLine - scrollOffset + 2 // +2 for header and padding

    if visualLine < 0 || visualLine >= len(lines) {
        // Cursor not visible, don't show dropdown
        return lines
    }

    // Insert dropdown lines after cursor line
    dropdownLines := strings.Split(dropdown, "\n")
    result := make([]string, 0, len(lines)+len(dropdownLines))
    result = append(result, lines[:visualLine+1]...)
    result = append(result, dropdownLines...)
    result = append(result, lines[visualLine+1:]...)

    return result
}
```
