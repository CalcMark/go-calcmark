  func TestPaneWidthRatio(t *testing.T) {
      // Verify 60/40 ratio calculation
      m := New(nil)
      m.width = 100

      sourceWidth, previewWidth := m.GetPaneWidths(100)

      if sourceWidth != 60 {
          t.Errorf("sourceWidth = %d, want 60", sourceWidth)
      }
      if previewWidth != 40 {
          t.Errorf("previewWidth = %d, want 40", previewWidth)
      }
  }

  func TestPreviewHeader(t *testing.T) {
      // Verify header shows "Results"
      m := New(nil)
      m.width = 100
      m.height = 24

      view := m.View()

      if !strings.Contains(view, "Results") {
          t.Error("Preview header should contain 'Results'")
      }
      if strings.Contains(view, "Preview") {
          t.Error("Preview header should not contain 'Preview'")
      }
  }

  func TestAnonymousCalculationFormat(t *testing.T) {
      // Test that anonymous calculations show "→ value" format
      m := New(nil)
      lr := LineResult{
          IsCalc: true,
          Value:  "4",
          VarName: "", // Anonymous
      }

      result := m.renderCalcLine(lr, 40)

      if !strings.Contains(result, "→") {
          t.Error("Anonymous calculation should contain arrow")
      }
      if strings.Contains(result, "#") {
          t.Error("Anonymous calculation should not contain placeholder")
      }
  }

  PREVIEW-02: Results vertically aligned with source calculation lines
  PREVIEW-03: Variable assignments display as `variable_name -> result`
  PREVIEW-04: Anonymous calculations display as `# -> result` (updated to `-> result`)

  ```go
  func TestPreviewRequirements(t *testing.T) {
      t.Run("PREVIEW-01: only calc results", func(t *testing.T) {
          // Create document with markdown and calc
          content := `# Heading
  a = 5`
          doc, _ := document.NewDocument(content)
          m := New(doc)

          results := m.GetLineResults()

          // Line 0 is markdown - should show blank in preview
          if results[0].IsCalc {
              t.Error("Markdown line should not be IsCalc")
          }

          // Line 1 is calc - should have result
          if !results[1].IsCalc {
              t.Error("Calc line should be IsCalc")
          }
      })

      t.Run("PREVIEW-02: vertical alignment", func(t *testing.T) {
          // Verify source line N aligns with preview line N
          m := New(nil)
          m.width = 100
          m.height = 24

          aligned := m.computeAlignedModelFresh(60, 40)

          // Same number of visual lines in both panes
          if len(aligned.SourceLines) != len(aligned.PreviewLines) {
              t.Errorf("Source (%d) and preview (%d) line counts differ",
                  len(aligned.SourceLines), len(aligned.PreviewLines))
          }
      })

      t.Run("PREVIEW-03: variable assignment format", func(t *testing.T) {
          lr := LineResult{IsCalc: true, VarName: "total", Value: "100"}
          m := New(nil)
          result := m.renderCalcLine(lr, 40)

          if !strings.Contains(result, "total") {
              t.Error("Should contain variable name")
          }
          if !strings.Contains(result, "→") {
              t.Error("Should contain arrow")
          }
          if !strings.Contains(result, "100") {
              t.Error("Should contain value")
          }
      })

      t.Run("PREVIEW-04: anonymous calc format", func(t *testing.T) {
          lr := LineResult{IsCalc: true, VarName: "", Value: "42"}
          m := New(nil)
          result := m.renderCalcLine(lr, 40)

          if !strings.Contains(result, "→") {
              t.Error("Anonymous calc should have arrow")
          }
          // Should NOT have a placeholder like "#"
      })

      t.Run("PREVIEW-05: blank for non-calc", func(t *testing.T) {
          lr := LineResult{IsCalc: false, Source: "# Heading"}
          m := New(nil)
          result := m.renderCalcLine(lr, 40)

          // For non-calc lines, renderCalcLine returns markdown
          // But in preview pane, text blocks show blank (handled by aligned.go)
          // This test verifies the LineResult structure
          if lr.Value != "" {
              t.Error("Non-calc line should have empty Value")
          }
      })
  }
  ```
