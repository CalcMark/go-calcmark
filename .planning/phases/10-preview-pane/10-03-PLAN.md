  ```go
  // currencySymbols maps common currency codes to their symbols
  var currencySymbols = map[string]string{
      "USD": "$",
      "GBP": "GBP",
      "JPY": "JPY",
      "CAD": "CAD",
      // EUR stays as EUR (symbol position varies by locale)
  }

  func FormatCurrency(c *types.Currency) string {
      if c == nil {
          return ""
      }

      // Normalize symbol (USD -> $)
      symbol := c.Symbol
      if normalized, ok := currencySymbols[symbol]; ok {
          symbol = normalized
      }

      absValue, _ := c.Value.Abs().Float64()
      isNegative := c.Value.IsNegative()

      var numStr string

      switch {
      case absValue < 10000:
          // Small/mid values: show with decimals and separators
          if absValue >= 1000 {
              // Use thousand separators
              numStr = formatCurrencyWithSeparators(c.Value.Abs())
          } else {
              // Standard 2 decimal format
              numStr = c.Value.Abs().StringFixed(2)
          }
      default:
          // Large values: use K/M/B suffixes
          numStr = formatNumberWithSuffix(c.Value.Abs())
      }

      // Build result with proper sign and symbol position
      if isNegative {
          return "-" + symbol + numStr
      }
      return symbol + numStr
  }

  func formatCurrencyWithSeparators(value decimal.Decimal) string {
      // Format with 2 decimal places, then add separators to integer part
      str := value.StringFixed(2)
      parts := strings.Split(str, ".")
      intPart := addThousandSeparators(parts[0])
      if len(parts) > 1 {
          return intPart + "." + parts[1]
      }
      return intPart
  }
  ```
