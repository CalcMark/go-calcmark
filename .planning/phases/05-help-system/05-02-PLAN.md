**2. Create `cmd/calcmark/tui/editor/help_overlay.go`:**

```go
package editor

import (
    "github.com/charmbracelet/bubbles/help"
    "github.com/charmbracelet/lipgloss"
)

// renderHelpOverlay renders a centered help panel showing all keybindings.
func (m Model) renderHelpOverlay() string {
    // Create help model
    h := help.New()
    h.ShowAll = true
    h.Width = m.width - 20 // Leave margins

    // Style the help content
    helpContent := h.View(m.keys)

    // Create bordered box
    boxStyle := lipgloss.NewStyle().
        Border(lipgloss.RoundedBorder()).
        BorderForeground(lipgloss.Color("62")).
        Padding(1, 2).
        Width(m.width - 10).
        MaxHeight(m.height - 4)

    title := lipgloss.NewStyle().
        Bold(true).
        Foreground(lipgloss.Color("212")).
        Render("CalcMark Help")

    footer := lipgloss.NewStyle().
        Faint(true).
        Render("Press F1 or Esc to close")

    content := lipgloss.JoinVertical(lipgloss.Left,
        title,
        "",
        helpContent,
        "",
        footer,
    )

    return boxStyle.Render(content)
}
```

    } else {

    }
    return m, nil

// If in help mode, only respond to escape to close

    return m, nil // Ignore other keys while help is shown
