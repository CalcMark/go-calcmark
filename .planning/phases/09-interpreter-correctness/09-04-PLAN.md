```go
func TestNaturalLanguageForms(t *testing.T) {
    tests := []struct {
        name     string
        nlForm   string  // Natural language form
        stdForm  string  // Standard form
        expected string
    }{
        // average of
        {"average of 3 numbers", "average of 1, 2, 3\n", "avg(1, 2, 3)\n", "2"},
        {"average of 5 numbers", "average of 10, 20, 30, 40, 50\n", "avg(10, 20, 30, 40, 50)\n", "30"},
        {"average of 2 numbers", "average of 100, 200\n", "avg(100, 200)\n", "150"},
        {"average of with decimals", "average of 1.5, 2.5, 3.0\n", "avg(1.5, 2.5, 3.0)\n", "2.333..."},

        // square root of
        {"square root of 25", "square root of 25\n", "sqrt(25)\n", "5"},
        {"square root of 100", "square root of 100\n", "sqrt(100)\n", "10"},
        {"square root of 2", "square root of 2\n", "sqrt(2)\n", "1.414..."},
        {"square root of 0", "square root of 0\n", "sqrt(0)\n", "0"},
    }

    for _, tc := range tests {
        t.Run(tc.name, func(t *testing.T) {
            // Evaluate natural language form
            nlResult := evaluate(tc.nlForm)

            // Evaluate standard form
            stdResult := evaluate(tc.stdForm)

            // Both should produce same result
            if nlResult != stdResult {
                t.Errorf("NL form %q produced %s, but std form %q produced %s",
                    tc.nlForm, nlResult, tc.stdForm, stdResult)
            }

            // Result should match expected
            if !matches(nlResult, tc.expected) {
                t.Errorf("Expected %s, got %s", tc.expected, nlResult)
            }
        })
    }
}
```

Verify equivalence between natural language and standard forms.
