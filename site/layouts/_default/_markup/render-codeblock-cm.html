{{- /* Render hook for ```cm fenced code blocks. {{- /* Render the source code with Hugo syntax highlighting */ -}} {{ transform.Highlight $raw "text" (dict "lineNos" false) }} {{- with $results -}} {{- if .error -}}
Error
{{ .error }}
{{- else -}} {{- /* Only show results div if there are actual results */ -}} {{- $hasResults := false -}} {{- range .lines -}} {{- if .result -}}{{ $hasResults = true }}{{- end -}} {{- end -}} {{- if $hasResults -}}
Results

{{ .source }} → {{ .result }}
